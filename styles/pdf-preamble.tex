% PDF-only production settings for the CRC print route.

% Improve code block formatting when fvextra is available. If the local
% TeX installation lacks fvextra, keep Pandoc's default Highlighting setup.
\IfFileExists{fvextra.sty}{%
  \usepackage{fvextra}
  \DefineVerbatimEnvironment{Highlighting}{Verbatim}{
    breaklines=true,
    breakanywhere=true,
    showtabs=false,
    showspaces=false,
    obeytabs=true,
    commandchars=\\\{\},
    breaksymbolleft={},
  }
  \fvset{breaksymbol={}}
}{}

% Keep selected section headings with enough following material to avoid
% isolated headings at the foot of a page.
\usepackage{needspace}

% Captions already include publisher-style figure numbers in the source
% files, e.g. "Figure 4.1." and "Table 4.1."; do not add LaTeX's
% automatic "Figure N:" or "Table N:" prefix.
\usepackage{caption}
\captionsetup[figure]{labelformat=empty,labelsep=none}
\captionsetup[table]{labelformat=empty,labelsep=none}

% Keep the title, contents, and complete Preface in Roman-numbered front
% matter. Quarto's book template calls \mainmatter before index.qmd, so hold
% that automatic switch until KOMA-Script has opened the first Part page.
\usepackage{etoolbox}
\makeatletter
\let\cssoriginalmainmatter\mainmatter
\renewcommand*{\mainmatter}{}
\AtBeginDocument{\frontmatter}
\newif\ifcssmainmatterstarted
\newcommand*{\cssstartmainmatter}[1]{%
  \ifstrequal{#1}{part}{%
    \ifcssmainmatterstarted\else
      \global\@mainmattertrue
      \pagenumbering{arabic}%
      \global\let\mainmatter\cssoriginalmainmatter
      \global\cssmainmatterstartedtrue
    \fi
  }{}%
}
\AddtoDoHook{heading/postinit}{\cssstartmainmatter}
\makeatother
